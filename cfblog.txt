GitHub Actions 가 매일 OpenAI 로 글을 만들어 커밋한다. 자동 생성 자체보다
"읽을 값이 없는 글이 대량으로 쌓이는 것"이 문제이므로, 그 지점을 막는 데
설계를 집중했다. 방금 걷어낸 /q 도어웨이 페이지와 같은 실패를 규모만 키워
반복하지 않기 위해서다.

형식과 검증
- 마크다운이 아니라 구조화된 JSON 으로 만든다. 비교 기준(왜 중요한지 +
  어떻게 확인하는지), 자주 하는 실수, 구매 전 체크리스트, FAQ 로 고정했다.
  형식이 정해져 있으면 얇은 글이 통과하기 어렵다.
- post-schema.ts 의 기준을 통과하지 못하면 저장하지 않는다. 항목 수와
  최소 분량을 모두 검사한다.
- "10만 명이 선택", "만족도 98%", "전문가 추천" 같은 확인 불가한 표현을
  정규식으로 잡아낸다. 자동 생성 글에서 가장 흔한 실패다. dB 같은 규격
  단위는 통과시킨다.
- 반려되면 이유를 그대로 모델에 돌려주고 한 번 더 시킨다. 두 번째도
  실패하면 그 주제를 건너뛴다. 기준을 낮춰 통과시키지 않는다.
- 화면에서 읽을 때도 다시 검증한다. 손으로 고치다 깨진 글이 공개되는 것을
  막는다. 저장소의 모든 글이 기준을 통과하는지 테스트로도 확인한다.

내용
- 주제는 무한 조합이 아니라 curated 목록이다. 실제 구매 고민 25개로 시작한다.
- 특정 제품을 밀지 않고 판단 기준만 다룬다. 읽고 나서 상세페이지를 직접
  비교할 수 있게 하는 것이 목적이다.
- AI 로 초안을 만든다는 사실을 글 하단에 밝힌다.
- 손으로 쓴 기준 글 한 편을 함께 넣었다. 생성 결과를 비교할 기준이 된다.

연결
- 푸터에서 링크한다. 어디에서도 링크되지 않으면 고아 문서가 된다. /q 가
  그랬다.
- sitemap 에는 검증을 통과해 실제로 공개되는 글만 담는다.

+
+on:
+  schedule:
+    # 매일 00:10 UTC (한국 09:10)

+  workflow_dispatch:
+    inputs:
+      count:
+        description: "이번 실행에서 생성할 글 수 (1~5)"
+        required: false
+        default: "3"
+
+permissions:
+  contents: write
+
+concurrency:
+  group: blog-generate
+  cancel-in-progress: false
+
+jobs:
+  generate:
+    runs-on: ubuntu-latest
+    steps:
+      - uses: actions/checkout@v4
+
+      - uses: actions/setup-node@v4
+        with:
+          node-version: "22"
+
+      - name: 글 생성
+        env:
+          OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}
+          POSTS_PER_RUN: ${{ github.event.inputs.count || '3' }}
+        run: node --experimental-strip-types scripts/generate-blog-posts.mts
+
+      # 형식이 깨진 글이 올라가면 화면에서 걸러지긴 하지만,

+async function callOpenAI(prompt: string): Promise<unknown> {
+  const apiKey = process.env.OPENAI_API_KEY?.trim();
+  if (!apiKey) throw new Error("OPENAI_API_KEY 가 설정되지 않았습니다.");
+
+  const response = await fetch("https://api.openai.com/v1/chat/completions", {
+    method: "POST",
+    headers: {
+      "Content-Type": "application/json",
+      Authorization: `Bearer ${apiKey}`,
+    },
+    body: JSON.stringify({
+      model: MODEL,
+      messages: [{ role: "user", content: prompt }],
+      response_format: { type: "json_object" },
+    }),
+  });
+
+  if (!response.ok) {
+    const detail = await response.text().catch(() => "");
+    throw new Error(`OpenAI 응답 실패 ${response.status}: ${detail.slice(0, 300)}`);
+  }
+
+  const payload = (await response.json()) as {
+    choices?: Array<{ message?: { content?: string } }>;
+  };
+  const content = payload.choices?.[0]?.message?.content;
+  if (!content) throw new Error("OpenAI 응답이 비어 있습니다.");
+  return JSON.parse(content);
+}

+async function loadWrittenSlugs(): Promise<Set<string>> {
+  if (!existsSync(POSTS_DIR)) return new Set();
+  const files = await readdir(POSTS_DIR);
+  return new Set(
+    files.filter((name) => name.endsWith(".json")).map((name) => name.slice(0, -5))
+  );
+}

+async function generateOne(topic: Topic): Promise<BlogPost | null> {
+  let problems: string[] | undefined;
+
+  // 한 번 반려되면 이유를 붙여 다시 시킨다. 두 번째도 실패하면 넘어간다.
+  for (let attempt = 1; attempt <= 2; attempt += 1) {
+    try {
+      const raw = await callOpenAI(buildPrompt(topic, problems));
+      const result = validateBlogPost({ ...(raw as object), slug: topic.slug });
+      if (result.ok) return result.post;
+
+      problems = result.problems;
+      console.warn(
+        `[blog] ${topic.slug} 시도 ${attempt} 반려:\n  - ${result.problems.join("\n  - ")}`
+      );
+    } catch (error) {
+      console.error(
+        `[blog] ${topic.slug} 시도 ${attempt} 오류:`,
+        error instanceof Error ? error.message : error
+      );
+      return null;
+    }
+  }
+  return null;
+}

+async function main() {
+  const topics = JSON.parse(await readFile(TOPICS_PATH, "utf8")) as Topic[];
+  const written = await loadWrittenSlugs();
+  const pending = topics.filter((topic) => !written.has(topic.slug));
+
+  if (pending.length === 0) {
+    console.log("[blog] 남은 주제가 없습니다. topics.json 에 주제를 추가하세요.");
+    return;
+  }
+
+  await mkdir(POSTS_DIR, { recursive: true });
+
+  let saved = 0;
+  for (const topic of pending.slice(0, POSTS_PER_RUN)) {
+    const post = await generateOne(topic);
+    if (!post) {
+      console.warn(`[blog] ${topic.slug} 건너뜀 (기준 미달)`);
+      continue;
+    }
+    await writeFile(
+      path.join(POSTS_DIR, `${post.slug}.json`),
+      `${JSON.stringify(post, null, 2)}\n`,
+      "utf8"
+    );
+    saved += 1;
+    console.log(`[blog] 저장: ${post.slug} — ${post.title}`);
+  }
+
+  console.log(
+    `[blog] 완료. ${saved}편 저장, 남은 주제 ${pending.length - Math.min(pending.length, POSTS_PER_RUN)}개`
+  );
+}
+
+main().catch((error) => {
+  console.error("[blog] 실행 실패:", error);
+  process.exit(1);
+});

+  }

+  }

+  const criteria = Array.isArray(raw.criteria)

+    }

+      );
+    }

+    : [];

+  }
+  if (mistakes.some((item) => item.length < POST_RULES.mistakeTextMin)) {

+      `mistakes의 각 항목은 최소 ${POST_RULES.mistakeTextMin}자여야 합니다.`

+  }

+  if (checklist.length < POST_RULES.checklistMin) {

+import { readdir, readFile } from "node:fs/promises";
+import { existsSync } from "node:fs";
+import path from "node:path";
+
+import { validateBlogPost, type BlogPost } from "@/lib/blog/post-schema";
+
+const POSTS_DIR = path.join(process.cwd(), "content/blog/posts");
+
+/**
+ * 저장된 글을 읽는다.
+ *
+ * 화면에서도 다시 검증한다. 생성 스크립트를 거치지 않고 파일이 들어오거나
+ * 손으로 고치다 형식이 깨지면, 반쯤 빈 글이 그대로 공개되기 때문이다.
+ */
+export async function getAllPosts(): Promise<BlogPost[]> {
+  if (!existsSync(POSTS_DIR)) return [];
+
+  const files = (await readdir(POSTS_DIR)).filter((name) =>
+    name.endsWith(".json")
+  );
+
+  const posts = await Promise.all(
+    files.map(async (name) => {
+      try {
+        const raw = JSON.parse(
+          await readFile(path.join(POSTS_DIR, name), "utf8")
+        );
+        const result = validateBlogPost(raw);
+        if (!result.ok) {
+          console.warn(`[blog] ${name} 형식 미달로 제외:`, result.problems);
+          return null;
+        }
+        return result.post;
+      } catch (error) {
+        console.warn(`[blog] ${name} 읽기 실패:`, error);
+        return null;
+      }
+    })
+  );
+
+  return posts
+    .filter((post): post is BlogPost => post !== null)
+    .sort((a, b) => b.publishedAt.localeCompare(a.publishedAt));
+}
+
+export async function getPostBySlug(slug: string): Promise<BlogPost | null> {
+  const posts = await getAllPosts();
+  return posts.find((post) => post.slug === slug) ?? null;
